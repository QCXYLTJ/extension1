写了份简短的代码:
for(var i in lib.character){
    let record=lib.config.qhly_winrecord[i];
    if(!record) continue;
    let recordMode=record.identity;
    if(!recordMode) continue;
    for(var j in recordMode){
        let win=recordMode[j].win||0;
        let lose=recordMode[j].lose||0;
        if(win+lose>=20){
            game.log(i,'('+i+')',j,' 赢:',win,' 输:',lose,' 胜率:',((win * 100 / (win + lose)).toFixed(2)) + "%");
        }
    }
}
